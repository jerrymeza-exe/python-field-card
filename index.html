def two_sum(nums, target):
    seen = {}                               # value -> index
    for i, x in enumerate(nums):
        if target - x in seen:

        seen[x] = i                         # store after checking
    return []

Valid Parentheses

Week 4
def is_valid(s):
    match = {")": "(", "]": "[", "}": "{"}
    stack = []
    for c in s:
        if c not in match:
            stack.append(c)                 # opening bracket
        elif not stack or stack.pop() != match[c]:
            return False
    return not stack

Daily Temperatures: monotonic stack

Week 4
def daily_temperatures(temps):
    answer = [0] * len(temps)
    stack = []                              # indexes still waiting for a warmer day
    for i, t in enumerate(temps):
        while stack and temps[stack[-1]] < t:
            j = stack.pop()
            answer[j] = i - j
        stack.append(i)
    return answer

Sliding Window Maximum: monotonic deque

Week 3
from collections import deque

def max_sliding_window(nums, k):
    dq = deque()                            # indexes; values decrease front to back
    out = []
    for i, x in enumerate(nums):
        while dq and nums[dq[-1]] <= x:
            dq.pop()
        dq.append(i)
        if dq[0] <= i - k:
            dq.popleft()                    # left the window
        if i >= k - 1:
            out.append(nums[dq[0]])
    return out

def find_kth_largest(nums, k):
    heap = []
    for x in nums:
        heapq.heappush(heap, x)
        if len(heap) > k:
            heapq.heappop(heap)             # drop the smallest
    return heap[0]                          # smallest of the k largest

Binary search

Week 5
def search(nums, target):
    lo, hi = 0, len(nums) - 1
    while lo <= hi:
        mid = (lo + hi) // 2                # Python ints never overflow
        if nums[mid] == target:
            return mid
        if nums[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1

Binary search on the answer: Koko

def min_eating_speed(piles, h):
    lo, hi = 1, max(piles)
    while lo < hi:
        mid = (lo + hi) // 2
        hours = sum((p + mid - 1) // mid for p in piles)   # ceiling division
        if hours <= h:
            hi = mid                        # fast enough; try slower
        else:
            lo = mid + 1
    return lo

def two_sum_sorted(numbers, target):
    l, r = 0, len(numbers) - 1
    while l < r:
        total = numbers[l] + numbers[r]
        if total == target:
            return [l + 1, r + 1]
        if total < target:
            l += 1                          # need a bigger sum
        else:
            r -= 1                          # need a smaller sum
    return []

Sliding window

Week 3
def length_of_longest_substring(s):
    window = set()
    l = best = 0
    for r, c in enumerate(s):
        while c in window:
            window.remove(s[l])             # shrink until valid
            l += 1
        window.add(c)
        best = max(best, r - l + 1)
    return best

from collections import defaultdict

def subarray_sum(nums, k):
    seen = defaultdict(int)
    seen[0] = 1                             # the empty prefix
    total = count = 0
    for x in nums:
        total += x
        count += seen[total - k]            # earlier prefixes that leave exactly k
        seen[total] += 1
    return count

Grid BFS: Rotting Oranges

Week 11
from collections import deque

def oranges_rotting(grid):
    rows, cols = len(grid), len(grid[0])
    q = deque()
    fresh = 0
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 2:
                q.append((r, c))            # multi-source: every rotten orange starts
            elif grid[r][c] == 1:
                fresh += 1
    minutes = 0
    while q and fresh:
        for _ in range(len(q)):             # one level = one minute
            r, c = q.popleft()
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = r + dr, c + dc
                if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == 1:
                    grid[nr][nc] = 2        # mark on push, not on pop
                    fresh -= 1
                    q.append((nr, nc))
        minutes += 1
    return minutes if fresh == 0 else -1

Grid DFS: Number of Islands

Week 10
def num_islands(grid):
    rows, cols = len(grid), len(grid[0])

    def dfs(r, c):
        if r < 0 or r >= rows or c < 0 or c >= cols or grid[r][c] != "1":
            return
        grid[r][c] = "0"                    # mark visited by sinking the land
        dfs(r + 1, c)
        dfs(r - 1, c)
        dfs(r, c + 1)
        dfs(r, c - 1)

    islands = 0
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == "1":
                islands += 1
                dfs(r, c)
    return islands

Topological sort: Course Schedule II

Weeks 13–14
from collections import deque

def find_order(n, prerequisites):
    adj = [[] for _ in range(n)]
    indegree = [0] * n
    for course, pre in prerequisites:       # take pre before course
        adj[pre].append(course)
        indegree[course] += 1
    q = deque(i for i in range(n) if indegree[i] == 0)
    order = []
    while q:
        u = q.popleft()
        order.append(u)
        for v in adj[u]:
            indegree[v] -= 1
            if indegree[v] == 0:
                q.append(v)
    return order if len(order) == n else []   # [] means there's a cycle

Union-find

Week 15
def count_components(n, edges):
    parent = list(range(n))
    size = [1] * n

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]   # path halving
            x = parent[x]
        return x

    components = n
    for a, b in edges:
        ra, rb = find(a), find(b)
        if ra == rb:
            continue                        # already connected: this edge makes a cycle
        if size[ra] < size[rb]:
            ra, rb = rb, ra                 # smaller tree goes under the larger
        parent[rb] = ra
        size[ra] += size[rb]
        components -= 1
    return components

Dijkstra: Network Delay Time

Week 16
import heapq
from collections import defaultdict

def network_delay_time(times, n, k):
    adj = defaultdict(list)
    for u, v, w in times:
        adj[u].append((v, w))
    dist = {}
    heap = [(0, k)]                         # (distance, node)
    while heap:
        d, u = heapq.heappop(heap)
        if u in dist:
            continue                        # already settled with a shorter distance
        dist[u] = d
        for v, w in adj[u]:
            if v not in dist:
                heapq.heappush(heap, (d + w, v))
    return max(dist.values()) if len(dist) == n else -1

def merge(intervals):
    intervals.sort()                        # by start, then end
    merged = []
    for start, end in intervals:
        if not merged or merged[-1][1] < start:
            merged.append([start, end])
        else:
            merged[-1][1] = max(merged[-1][1], end)
    return merged

Backtracking: Subsets

Week 18
def subsets(nums):
    result, path = [], []

    def backtrack(start):
        result.append(path[:])             # copy! path keeps changing
        for i in range(start, len(nums)):
            path.append(nums[i])            # choose
            backtrack(i + 1)                # explore
            path.pop()                      # un-choose

    backtrack(0)
    return result

DP bottom-up table: Longest Common Subsequence

Week 25
def longest_common_subsequence(a, b):
    dp = [[0] * (len(b) + 1) for _ in range(len(a) + 1)]
    for i in range(len(a) - 1, -1, -1):
        for j in range(len(b) - 1, -1, -1):
            if a[i] == b[j]:
                dp[i][j] = 1 + dp[i + 1][j + 1]
            else:
                dp[i][j] = max(dp[i + 1][j], dp[i][j + 1])
    return dp[0][0]

Trie of nested dicts

Week 27
class Trie:
    def __init__(self):
        self.root = {}

    def insert(self, word):
        node = self.root
        for c in word:
            node = node.setdefault(c, {})
        node["$"] = True                    # end-of-word marker

    def search(self, word, prefix=False):
        node = self.root
        for c in word:
            if c not in node:
                return False
            node = node[c]
        return prefix or "$" in node

def reverse_list(head):
    prev, curr = None, head
    while curr:
        nxt = curr.next                     # save before overwriting
        curr.next = prev
        prev = curr
        curr = nxt
    return prev

def has_cycle(head):
    slow = fast = head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow is fast:
            return True                     # Linked List Cycle
    return False                            # with no cycle, slow stops at the middle

            level.append(node.val)
            if node.left:
                q.append(node.left)
            if node.right:
                q.append(node.right)
        levels.append(level)
    return levels

nonlocal: Diameter of Binary Tree

Week 7
def diameter_of_binary_tree(root):
    best = 0

    def height(node):
        nonlocal best                       # we reassign best below
        if not node:
            return 0
        left, right = height(node.left), height(node.right)
        best = max(best, left + right)
        return 1 + max(left, right)

    height(root)
    return best

LRU cache

Week 21

Interviewers may ask for the version built from a dict plus a doubly linked list. OrderedDict is that structure, already written.

from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.items = OrderedDict()

    def get(self, key: int) -> int:
        if key not in self.items:
            return -1
        self.items.move_to_end(key)         # mark as most recently used
        return self.items[key]

    def put(self, key: int, value: int) -> None:
        self.items[key] = value
        self.items.move_to_end(key)
        if len(self.items) > self.capacity:
            self.items.popitem(last=False)  # evict the least recently used

Rate limiter: sliding window log

Week 22
import time
from collections import deque

class RateLimiter:
    """Allow at most `limit` requests per `window` seconds."""

    def __init__(self, limit: int, window: float, clock=time.monotonic):
        self.limit = limit
        self.window = window
        self.clock = clock                  # injectable, so tests can fake time
        self.calls = deque()

    def allow(self) -> bool:
        now = self.clock()
        while self.calls and now - self.calls[0] >= self.window:
            self.calls.popleft()            # forget calls outside the window
        if len(self.calls) < self.limit:
            self.calls.append(now)
            return True
        return False

Pub/sub with an abstract interface

Week 26
from abc import ABC, abstractmethod
from collections import defaultdict

class Subscriber(ABC):
    @abstractmethod
    def notify(self, topic: str, message: dict) -> None: ...

class EventBus:
    def __init__(self):
        self.subscribers = defaultdict(list)    # topic -> subscribers

    def subscribe(self, topic: str, sub: Subscriber) -> None:
        self.subscribers[topic].append(sub)

    def publish(self, topic: str, message: dict) -> None:
        for sub in self.subscribers.get(topic, []):   # get() doesn't add the topic
            sub.notify(topic, message)

Priority scheduler: dataclass in a heap

Week 28
import heapq
from dataclasses import dataclass, field
from typing import Optional

@dataclass(order=True)
class Task:
    priority: int                           # lower number runs first
    seq: int                                # tie-breaker: first in, first out
    name: str = field(compare=False)

class Scheduler:
    def __init__(self):
        self.heap = []
        self.count = 0

    def submit(self, name: str, priority: int) -> None:
        heapq.heappush(self.heap, Task(priority, self.count, name))
        self.count += 1

    def next_task(self) -> Optional[str]:
        return heapq.heappop(self.heap).name if self.heap else None

Producer and consumer threads

Concurrency

In standard CPython, threads take turns running Python code (the GIL). They help while waiting on I/O, not with CPU-heavy work.

import queue
import threading

def run_pipeline(readings):
    events = queue.Queue()                  # already thread-safe
    results = []

    def sensor():
        for r in readings:
            events.put(r)
        events.put(None)                    # sentinel: no more data

    def station():
        while (item := events.get()) is not None:   # blocks until an item arrives
            results.append(item)

    workers = [threading.Thread(target=sensor), threading.Thread(target=station)]
    for w in workers:
        w.start()
    for w in workers:
        w.join()
    return results
